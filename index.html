    function formatMoney(amount) {

    // Set Global Multiplier (1, 10, 100, 1000...)
    function setGlobalMultiplier(value) {
      globalMultiplier = value;
      const buttons = document.querySelectorAll('#preset-buttons button');
      buttons.forEach(btn => {
        btn.classList.remove('bg-blue-600', 'text-white');
        btn.classList.add('text-slate-400');
      });

      const activeIndex = [1, 10, 100, 1000, 10000].indexOf(value);
      if (activeIndex !== -1 && buttons[activeIndex]) {
        buttons[activeIndex].classList.add('bg-blue-600', 'text-white');
        buttons[activeIndex].classList.remove('text-slate-400');
      }

      renderGrid();
    }

    function getProcessedItems() {
      const searchTerm = document.getElementById('search-input').value.trim().toLowerCase();
      const sortOption = document.getElementById('sort-select').value;

      let result = itemsData.filter(item => item.name.toLowerCase().includes(searchTerm));

      switch (sortOption) {
        case 'price-asc': result.sort((a, b) => a.price - b.price); break;
        case 'price-desc': result.sort((a, b) => b.price - a.price); break;
        case 'count-desc': result.sort((a, b) => inventory[b.id] - inventory[a.id]); break;
        case 'name': result.sort((a, b) => a.name.localeCompare(b.name, 'ko')); break;
        case 'id-asc': default: result.sort((a, b) => a.id - b.id); break;
      }

      return result;
    }

    function handleDirectInputChange(itemId, value) {
      const num = parseInt(value, 10);
      if (isNaN(num) || num < 1) {
        document.getElementById(`input-${itemId}`).value = globalMultiplier;
      }
    }

    function buyItem(id) {
      const item = itemsData.find(i => i.id === id);
      const inputEl = document.getElementById(`input-${id}`);
      let qty = parseInt(inputEl ? inputEl.value : globalMultiplier, 10) || 1;

      const totalCost = item.price * qty;
      if (currentBalance < totalCost) {
        const maxAffordable = Math.floor(currentBalance / item.price);
        if (maxAffordable > 0) {
          qty = maxAffordable;
        } else {
          return;
        }
      }

      currentBalance -= item.price * qty;
      inventory[id] += qty;
      updateUI();
    }

    function sellItem(id) {
      const item = itemsData.find(i => i.id === id);
      const inputEl = document.getElementById(`input-${id}`);
      let qty = parseInt(inputEl ? inputEl.value : globalMultiplier, 10) || 1;

      if (inventory[id] < qty) {
        qty = inventory[id];
      }

      if (qty <= 0) return;

      currentBalance += item.price * qty;
      inventory[id] -= qty;
      updateUI();
    }

    function resetAll() {
      if (confirm("모든 구매 내역을 초기화하고 $200B 잔액으로 되돌리시겠습니까?")) {
        currentBalance = STARTING_BALANCE;
        itemsData.forEach(item => { inventory[item.id] = 0; });
        updateUI();
      }
    }

    function updateUI() {
      const balanceEl = document.getElementById('balance-display');
      balanceEl.textContent = formatMoney(currentBalance);

      const percent = (currentBalance / STARTING_BALANCE) * 100;
      const progressBar = document.getElementById('progress-bar');
      const percentEl = document.getElementById('percent-display');
      
      progressBar.style.width = `${Math.max(0, percent)}%`;
      percentEl.textContent = `${percent.toFixed(2)}%`;

      if (percent < 10) {
        balanceEl.className = "text-2xl sm:text-4xl font-extrabold text-rose-500 balance-glow transition-all duration-300";
        progressBar.className = "bg-rose-500 h-2.5 rounded-full transition-all duration-300";
      } else if (percent < 50) {
        balanceEl.className = "text-2xl sm:text-4xl font-extrabold text-amber-400 balance-glow transition-all duration-300";
        progressBar.className = "bg-gradient-to-r from-amber-500 to-rose-500 h-2.5 rounded-full transition-all duration-300";
      } else {
        balanceEl.className = "text-2xl sm:text-4xl font-extrabold text-emerald-400 balance-glow transition-all duration-300";
        progressBar.className = "bg-gradient-to-r from-emerald-500 to-blue-500 h-2.5 rounded-full transition-all duration-300";
      }

      renderGrid();
      renderReceipt();
    }

    function renderReceipt() {
      const receiptContainer = document.getElementById('receipt-container');
      const totalSpentEl = document.getElementById('total-spent-display');

      const purchasedItems = itemsData.filter(item => inventory[item.id] > 0);
      const totalSpent = STARTING_BALANCE - currentBalance;

      totalSpentEl.textContent = formatMoney(totalSpent);

      if (purchasedItems.length === 0) {
        receiptContainer.innerHTML = `<p class="text-slate-500 text-center py-8 text-sm">아직 지른 물품이 없습니다. 마음껏 쇼핑해보세요!</p>`;
        return;
      }

      receiptContainer.innerHTML = `
        <div class="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-3">
          ${purchasedItems.map(item => {
            const count = inventory[item.id];
            const itemTotal = item.price * count;
            return `
              <div class="bg-slate-800/60 border border-slate-700/60 rounded-xl p-3 flex items-center justify-between">
                <div class="flex items-center gap-2.5">
                  <span class="text-2xl">${item.emoji}</span>
                  <div>
                    <div class="text-xs font-bold text-slate-200">${item.name}</div>
                    <div class="text-[11px] text-slate-400">${formatMoney(item.price)} × ${count.toLocaleString()}개</div>
                  </div>
                </div>
                <div class="text-right">
                  <div class="text-xs font-extrabold text-blue-400">${formatMoney(itemTotal)}</div>
                </div>
              </div>
            `;
          }).join('')}
        </div>
      `;
    }
